export const copyJs = `
var t = this.getAttribute('data-copy') || '';
var s = this.querySelectorAll('svg');
var b = this;
if (navigator.clipboard && window.isSecureContext) {
  navigator.clipboard.writeText(t).catch(function () {});
} else {
  var a = document.createElement('textarea');
  a.value = t;
  a.setAttribute('readonly', '');
  a.style.cssText = 'position:fixed;top:0;left:-9999px;opacity:0';
  document.body.appendChild(a);
  a.select();
  try { document.execCommand('copy'); } catch (e) {}
  document.body.removeChild(a);
}
if (s[1]) {
  s[0].classList.add('hidden');
  s[1].classList.remove('hidden');
  setTimeout(function () {
    b.querySelectorAll('svg')[0].classList.remove('hidden');
    b.querySelectorAll('svg')[1].classList.add('hidden');
  }, 1500);
}
var l = this.querySelector('[data-copy-label]');
if (l) {
  if (!l.dataset.o) l.dataset.o = l.textContent;
  l.textContent = '已复制！';
  clearTimeout(this._lt);
  this._lt = setTimeout(function () { l.textContent = l.dataset.o; }, 1500);
}
`;
